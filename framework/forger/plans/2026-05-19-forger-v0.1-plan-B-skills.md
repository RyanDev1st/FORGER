# FORGER v0.1 — Plan B: Phase Skills + Lane Mandates

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans`. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Author the 7 phase SKILL.md files + 3 FIND lane mandates + supporting refs/examples. Each phase is independently invocable by an agent and produces artifacts that validate against the schemas built in Plan A.

**Architecture:** Skill markdown files (no line cap per CLAUDE.md decision D5). Each phase SKILL.md follows the uniform template from DESIGN.md §6: frontmatter, Identity, Inputs, Outputs, Gates, Procedure, Exit. Lane mandates follow §7. Refs are non-skill markdown — concrete calibration / pattern docs.

**Tech Stack:** Markdown only. Validation = read each example artifact produced by procedure and confirm it validates against the schema via `_lib/ledger.mjs`.

**Demo at end of Plan B:**
- Each `phases/<phase>/SKILL.md` exists, has correct frontmatter, references the right gates/tools/schemas.
- `dev/scripts/validate_all_examples.mjs` validates every `phases/*/examples/*.yaml` against its schema.
- `dev/scripts/check_skill_links.mjs` confirms every cross-reference (`refs/...`, `gates/...`, `tools/...`, `schemas/...`) resolves to an existing file.

**Depends on:** Plan A (schemas, _lib, gates, tools, hooks all in place).

- [ ] **Step 1: Implement**

```js
#!/usr/bin/env node
import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import {
  readYaml,
  validateDoW, validateSourceEntry, validateClaimEntry,
  validateRiskMap, validateProbeResult, validateRetroNote,
  validateFailureHypothesis,
} from '../../_lib/ledger.mjs';

const here = path.dirname(fileURLToPath(import.meta.url));
const root = path.resolve(here, '..', '..');

// Mapping path-tail substring -> validator
const checks = [
  { match: 'dow.familiar-domain.yaml',          fn: validateDoW,                kind: 'object' },
  { match: 'dow.novel-domain.yaml',             fn: validateDoW,                kind: 'object' },
  { match: 'risk_map.example.yaml',             fn: validateRiskMap,            kind: 'object' },
  { match: 'retro_note.example.yaml',           fn: validateRetroNote,          kind: 'object' },
  { match: 'source_ledger.example.yaml',        fn: validateSourceEntry,        kind: 'array'  },
  { match: 'claim_ledger.example.yaml',         fn: validateClaimEntry,         kind: 'array'  },
  { match: 'failure_hypotheses.example.yaml',   fn: validateFailureHypothesis,  kind: 'array'  },
];

function walk(dir) {
  const out = [];
  for (const e of fs.readdirSync(dir, { withFileTypes: true })) {
    const p = path.join(dir, e.name);
    if (e.isDirectory()) out.push(...walk(p));
    else if (p.endsWith('.yaml')) out.push(p);
  }
  return out;
}

const examplesRoot = path.join(root, 'phases');
let allOk = true;
for (const file of walk(examplesRoot).filter(f => f.includes(`${path.sep}examples${path.sep}`))) {
  const check = checks.find(c => file.endsWith(c.match));
  if (!check) {
    console.log(`skip ${file} (no validator mapped)`);
    continue;
  }
  const data = readYaml(file);
  if (check.kind === 'object') {
    const r = check.fn(data);
    if (!r.valid) { allOk = false; console.error(`FAIL ${file}: ${JSON.stringify(r.errors)}`); }
    else console.log(`ok   ${file}`);
  } else {
    if (!Array.isArray(data)) { allOk = false; console.error(`FAIL ${file}: expected array`); continue; }
    let ok = true;
    data.forEach((item, i) => {
      const r = check.fn(item);
      if (!r.valid) { ok = false; console.error(`FAIL ${file}[${i}]: ${JSON.stringify(r.errors)}`); }
    });
    if (ok) console.log(`ok   ${file} (${data.length} items)`);
    else allOk = false;
  }
}

process.exit(allOk ? 0 : 1);
```

```js
#!/usr/bin/env node
import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const here = path.dirname(fileURLToPath(import.meta.url));
const root = path.resolve(here, '..', '..');

// Match relative refs: `refs/...`, `gates/...`, `tools/...`, `schemas/...`,
// `templates/...`, `lanes/...`, `_lib/...`, `hooks/...`.
const LINK_RE = /\b((refs|gates|tools|schemas|templates|lanes|_lib|hooks)\/[A-Za-z0-9_./-]+\.(md|mjs|yaml|jsonl|json|txt))\b/g;

function walk(dir, accept = () => true) {
  const out = [];
  for (const e of fs.readdirSync(dir, { withFileTypes: true })) {
    const p = path.join(dir, e.name);
    if (e.isDirectory()) {
      if (e.name === 'node_modules' || e.name === 'workspaces') continue;
      out.push(...walk(p, accept));
    } else if (accept(p)) out.push(p);
  }
  return out;
}

const skillFiles = walk(path.join(root, 'phases'), p =>
  p.endsWith('SKILL.md') || p.endsWith('.md'));
skillFiles.push(path.join(root, 'SKILL.md'));

let allOk = true;
for (const sf of skillFiles) {
  if (!fs.existsSync(sf)) continue;
  const text = fs.readFileSync(sf, 'utf8');
  const seen = new Set();
  for (const m of text.matchAll(LINK_RE)) {
    const rel = m[1];
    if (seen.has(rel)) continue;
    seen.add(rel);

    // Resolve against multiple plausible roots:
    // 1) phase-local (sibling of skill file)
    // 2) plugin root
    const candidates = [
      path.resolve(path.dirname(sf), rel),
      path.resolve(root, rel),
    ];
    if (!candidates.some(p => fs.existsSync(p))) {
      console.error(`BROKEN ${sf}: ${rel}`);
      allOk = false;
    }
  }
}

if (allOk) console.log('all skill links resolve');
process.exit(allOk ? 0 : 1);
```
